# frontends (CLI + server) for one backend library
function(qe_frontends suffix backend)
  add_executable(qwen35-engine${suffix} src/main.cpp)
  target_link_libraries(qwen35-engine${suffix} PRIVATE ${backend})
  target_compile_options(qwen35-engine${suffix} PRIVATE ${QE_CPU_FLAGS})
  add_executable(qwen35-server${suffix} src/server.cpp)
  target_include_directories(qwen35-server${suffix} PRIVATE ${CMAKE_BINARY_DIR}/gen)
  target_link_libraries(qwen35-server${suffix} PRIVATE ${backend})
  if(WIN32)  # Winsock for the server's HTTP sockets
    target_link_libraries(qwen35-server${suffix} PRIVATE ws2_32)
  endif()
  target_compile_options(qwen35-server${suffix} PRIVATE ${QE_CPU_FLAGS})
  install(TARGETS qwen35-engine${suffix} qwen35-server${suffix} RUNTIME DESTINATION bin)
endfunction()

# ---- Vulkan backend: GLSL compute shaders compiled to SPIR-V at build time and embedded
if(QE_VULKAN)
  find_package(Vulkan REQUIRED)
  find_program(GLSLC glslc REQUIRED)
  file(GLOB QE_VK_SHADERS ${CMAKE_SOURCE_DIR}/src/vk/*.comp)
  file(GLOB QE_VK_INCLUDES ${CMAKE_SOURCE_DIR}/src/vk/*.glsl)
  # the variant list is read from each shader at configure time
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${QE_VK_SHADERS})
  # one SPIR-V module per (shader, variant); variants are listed in the shader's first line:
  #   // variants: NAME:DEFINES;NAME2:DEFINES ...
  set(QE_SPV_HEADERS)
  foreach(sh ${QE_VK_SHADERS})
    get_filename_component(base ${sh} NAME_WE)
    file(STRINGS ${sh} first LIMIT_COUNT 1)
    set(variants "${base}:")
    if(first MATCHES "^// variants: (.*)$")
      set(variants "${CMAKE_MATCH_1}")
    endif()
    string(REPLACE " " ";" variants "${variants}")
    foreach(v ${variants})
      string(REPLACE ":" ";" parts "${v}")
      list(GET parts 0 vname)
      list(LENGTH parts np)
      set(defs)
      if(np GREATER 1)
        list(GET parts 1 dstr)
        string(REPLACE "," ";" dlist "${dstr}")
        foreach(d ${dlist})
          if(d)
            list(APPEND defs -D${d})
          endif()
        endforeach()
      endif()
      set(spv ${CMAKE_BINARY_DIR}/spv/${vname}.spv)
      set(hdr ${CMAKE_BINARY_DIR}/spv/${vname}.h)
      add_custom_command(OUTPUT ${hdr}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/spv
        # vulkan1.3 target env: the Windows SDK glslc rejects GL_EXT_integer_dot_product
        # (mvc.comp I8 variants) under vulkan1.2, where it is core in 1.3 / SPIR-V 1.6.
        COMMAND ${GLSLC} --target-env=vulkan1.3 -O -I${CMAKE_SOURCE_DIR}/src/vk ${defs} -o ${spv} ${sh}
        COMMAND ${CMAKE_COMMAND} -DIN=${spv} -DOUT=${hdr} -DNAME=spv_${vname} -P ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
        DEPENDS ${sh} ${QE_VK_INCLUDES} ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
        COMMENT "glslc ${vname}")
      list(APPEND QE_SPV_HEADERS ${hdr})
      list(APPEND QE_SPV_NAMES ${vname})
    endforeach()
  endforeach()
  set(reg "#pragma once\n")
  foreach(n ${QE_SPV_NAMES})
    string(APPEND reg "#include \"${n}.h\"\n")
  endforeach()
  string(APPEND reg "struct SpvEntry { const char* name; const unsigned* code; unsigned size; };\nstatic const SpvEntry kSpv[] = {\n")
  foreach(n ${QE_SPV_NAMES})
    string(APPEND reg "  {\"${n}\", spv_${n}, sizeof(spv_${n})},\n")
  endforeach()
  string(APPEND reg "};\n")
  file(WRITE ${CMAKE_BINARY_DIR}/spv/spv_all.h.in "${reg}")
  configure_file(${CMAKE_BINARY_DIR}/spv/spv_all.h.in ${CMAKE_BINARY_DIR}/spv/spv_all.h COPYONLY)
  add_custom_target(qe_spv DEPENDS ${QE_SPV_HEADERS})

  add_library(qevk STATIC src/vk_model.cpp src/vk_ctx.cpp src/backend_vk.cpp)
  add_dependencies(qevk qe_spv)
  target_include_directories(qevk PUBLIC src PRIVATE ${CMAKE_BINARY_DIR}/spv)
  target_compile_options(qevk PRIVATE ${QE_CPU_FLAGS} -Wall -Wno-unused-function)
  target_link_libraries(qevk PUBLIC qecommon Vulkan::Vulkan)
  qe_frontends("-vk" qevk)
  if(NOT QE_CUDA)  # CPU kernel test (the CUDA build has it among its tools)
    add_executable(qe_test_cpu tools/qe_test_cpu.cpp)
    target_link_libraries(qe_test_cpu PRIVATE qecommon)
    target_compile_options(qe_test_cpu PRIVATE ${QE_CPU_FLAGS})
    install(TARGETS qe_test_cpu RUNTIME DESTINATION bin)
  endif()
  add_executable(qe_vk_test tools/qe_vk_test.cpp)
  target_include_directories(qe_vk_test PRIVATE ${CMAKE_BINARY_DIR}/spv)
  target_link_libraries(qe_vk_test PRIVATE qevk)
  target_compile_options(qe_vk_test PRIVATE ${QE_CPU_FLAGS})
  install(TARGETS qe_vk_test RUNTIME DESTINATION bin)
endif()

  set_source_files_properties(src/model.cpp src/kernels.cu src/backend_hip.cpp PROPERTIES LANGUAGE HIP)

  target_compile_definitions(qehip PUBLIC QE_HIP=1)
